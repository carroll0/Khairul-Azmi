    <script>
        const messyData = [
            { id: '001', name: 'john doe', dept: 'hr', salary: '5000', date: '1/15/2024', status: 'active' },
            { id: '2', name: 'JANE SMITH', dept: 'IT', salary: '6500.00', date: '15-Feb-2024', status: 'Active' },
            { id: '03', name: 'Bob Johnson', dept: 'finance', salary: '5500', date: '2024/03/10', status: 'ACTIVE' },
            { id: '4', name: 'alice brown', dept: 'HR', salary: '4800.5', date: '04/01/24', status: 'active' },
            { id: '005', name: 'Charlie Davis', dept: 'it', salary: '7000', date: '5-15-2024', status: 'Active' },
            { id: '6', name: 'eve WILSON', dept: 'Finance', salary: '5200', date: '2024-06-20', status: 'ACTIVE' },
        ];

        const cleanData = [
            { id: 'EMP-001', name: 'John Doe', dept: 'Human Resources', salary: '$5,000.00', date: '2024-01-15', status: 'Active' },
            { id: 'EMP-002', name: 'Jane Smith', dept: 'Information Technology', salary: '$6,500.00', date: '2024-02-15', status: 'Active' },
            { id: 'EMP-003', name: 'Bob Johnson', dept: 'Finance', salary: '$5,500.00', date: '2024-03-10', status: 'Active' },
            { id: 'EMP-004', name: 'Alice Brown', dept: 'Human Resources', salary: '$4,800.50', date: '2024-04-01', status: 'Active' },
            { id: 'EMP-005', name: 'Charlie Davis', dept: 'Information Technology', salary: '$7,000.00', date: '2024-05-15', status: 'Active' },
            { id: 'EMP-006', name: 'Eve Wilson', dept: 'Finance', salary: '$5,200.00', date: '2024-06-20', status: 'Active' },
        ];

        let isProcessed = false;

        function renderTable(data) {
            const tbody = document.getElementById('tableBody');
            tbody.innerHTML = '';
            
            data.forEach((row, idx) => {
                const tr = document.createElement('tr');
                tr.className = `border-b transition-all duration-500 ${
                    isProcessed ? 'bg-green-50 hover:bg-green-100' : 'bg-red-50 hover:bg-red-100'
                } fade-in`;
                tr.style.animationDelay = `${idx * 0.1}s`;
                
                tr.innerHTML = `
                    <td class="px-4 py-3 font-mono ${isProcessed ? 'text-green-800 font-semibold' : 'text-red-600'}">${row.id}</td>
                    <td class="px-4 py-3 ${isProcessed ? 'font-semibold' : ''}">${row.name}</td>
                    <td class="px-4 py-3">${row.dept}</td>
                    <td class="px-4 py-3 font-mono">${row.salary}</td>
                    <td class="px-4 py-3 font-mono text-sm">${row.date}</td>
                    <td class="px-4 py-3">
                        <span class="px-3 py-1 rounded-full text-xs font-semibold ${
                            isProcessed ? 'bg-green-200 text-green-800' : 'bg-gray-200 text-gray-700'
                        }">
                            ${row.status}
                        </span>
                    </td>
                `;
                
                tbody.appendChild(tr);
            });
        }

        function updateSummary() {
            const summaryBox = document.getElementById('summaryBox');
            
            if (!isProcessed) {
                summaryBox.innerHTML = `
                    <div class="bg-red-50 border-2 border-red-200 rounded-xl p-6 fade-in">
                        <h3 class="font-bold text-red-800 mb-3 text-xl">⚠ Data Quality Issues Detected:</h3>
                        <ul class="text-red-700 space-y-2 ml-6 list-disc">
                            <li>Inconsistent ID formatting (001, 2, 03, 4, 005, 6)</li>
                            <li>Mixed case names (lowercase, UPPERCASE, Title Case, mixed)</li>
                            <li>Inconsistent department names (hr vs HR vs Human Resources)</li>
                            <li>Inconsistent salary formatting (5000 vs 6500.00 vs $X,XXX.XX)</li>
                            <li>Multiple date formats (1/15/2024, 15-Feb-2024, 2024/03/10, 04/01/24)</li>
                            <li>Inconsistent status capitalization (active vs Active vs ACTIVE)</li>
                        </ul>
                    </div>
                `;
            } else {
                summaryBox.innerHTML = `
                    <div class="bg-green-50 border-2 border-green-200 rounded-xl p-6 fade-in">
                        <h3 class="font-bold text-green-800 mb-3 text-xl">✓ Automation Completed Successfully!</h3>
                        <ul class="text-green-700 space-y-2 ml-6 list-disc">
                            <li>✓ Standardized all employee IDs to EMP-XXX format</li>
                            <li>✓ Applied proper name capitalization (Title Case)</li>
                            <li>✓ Expanded department abbreviations to full names</li>
                            <li>✓ Formatted all salaries with $ symbol and thousand separators</li>
                            <li>✓ Standardized all dates to YYYY-MM-DD ISO format</li>
                            <li>✓ Normalized status field to consistent "Active" format</li>
                        </ul>
                    </div>
                `;
            }
        }

        function processData() {
            const statusBadge = document.getElementById('statusBadge');
            const processBtn = document.getElementById('processBtn');
            const fileName = document.getElementById('fileName');
            
            statusBadge.innerHTML = '<div class="flex items-center gap-2"><div class="animate-spin rounded-full h-5 w-5 border-b-2 border-yellow-800"></div> Processing data...</div>';
            statusBadge.className = 'bg-yellow-100 text-yellow-800 px-6 py-3 rounded-full font-semibold text-lg processing';
            
            processBtn.disabled = true;
            processBtn.classList.add('opacity-50', 'cursor-not-allowed');
            
            setTimeout(() => {
                isProcessed = true;
                renderTable(cleanData);
                updateSummary();
                
                statusBadge.innerHTML = '✓ Data Cleaned & Structured';
                statusBadge.className = 'bg-green-100 text-green-800 px-6 py-3 rounded-full font-semibold text-lg fade-in';
                
                fileName.textContent = 'employee_data_CLEANED.xlsx';
                
                document.getElementById('processBtn').style.display = 'none';
                document.getElementById('resetBtn').style.display = 'inline-block';
            }, 1500);
        }

        function resetData() {
            isProcessed = false;
            renderTable(messyData);
            updateSummary();
            
            const statusBadge = document.getElementById('statusBadge');
            statusBadge.innerHTML = '⚠ Raw Messy Data';
            statusBadge.className = 'bg-red-100 text-red-800 px-6 py-3 rounded-full font-semibold text-lg fade-in';
            
            document.getElementById('fileName').textContent = 'employee_data_raw.xlsx';
            
            const processBtn = document.getElementById('processBtn');
            processBtn.style.display = 'inline-block';
            processBtn.disabled = false;
            processBtn.classList.remove('opacity-50', 'cursor-not-allowed');
            
            document.getElementById('resetBtn').style.display = 'none';
        }

        // Initialize
        renderTable(messyData);
        updateSummary();
    </script>
